function parseLine(line) { let parts = line.split(/\t+|\s+[–—-]\s+|\s*[:;|]\s*/); if (parts.length < 2) { parts = line.split(/\s*[–—-]\s*/); } if (parts.length < 2) { const match = line.trim().match(/^(\S+)\s+(.+)$/); if (!match) return [line.trim(), ""]; parts = [match[1], match[2]]; } let left = parts.shift().trim(); let right = parts.join(" - ").trim(); function removePronunciation(text) { return text .replace(/\[[^\]]*\]/g, " ") .replace(/\/[^/]+\//g, " ") .replace(/\s+/g, " ") .trim(); } left = removePronunciation(left); right = removePronunciation(right); const order = document.querySelector("#word-order").value; const polishCharacters = /[ąćęłńóśźż]/i; if (order === "pl-first") return [right, left]; if (order === "en-first") return [left, right]; if (polishCharacters.test(left) && !polishCharacters.test(right)) { return [right, left]; } return [left, right]; }
